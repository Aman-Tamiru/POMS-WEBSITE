"use client";

import React, { createContext, useContext, useEffect, useState } from "react";
import en from "@/locales/en.json";
import am from "@/locales/am.json";
import so from "@/locales/so.json";
import zhCN from "@/locales/zh-CN.json";
import hi from "@/locales/hi.json";
import es from "@/locales/es.json";

export type SupportedLanguage = "en" | "am" | "so" | "zh-CN" | "hi" | "es";

export interface LanguageContextType {
  language: SupportedLanguage;
  setLanguage: (lang: SupportedLanguage) => void;
  t: (keyPath: string) => string;
}

type Dictionary = {
  [key: string]: string | Dictionary;
};

const dictionaries: Record<SupportedLanguage, Dictionary> = {
  en,
  am,
  so,
  "zh-CN": zhCN,
  hi,
  es,
};

export const LANGUAGE_LABELS: Record<SupportedLanguage, { label: string; code: string; flag: string }> = {
  en: { label: "English", code: "EN", flag: "🇬🇧" },
  am: { label: "አማርኛ", code: "AM", flag: "🇪🇹" },
  so: { label: "Soomaali", code: "SO", flag: "🇸🇴" },
  "zh-CN": { label: "中文", code: "ZH", flag: "🇨🇳" },
  hi: { label: "हिन्दी", code: "HI", flag: "🇮🇳" },
  es: { label: "Español", code: "ES", flag: "🇪🇸" },
};

const LanguageContext = createContext<LanguageContextType>({
  language: "en",
  setLanguage: () => {},
  t: (key) => key,
});

export function LanguageProvider({ children }: { children: React.ReactNode }) {
  const [language, setLanguageState] = useState<SupportedLanguage>("en");

  useEffect(() => {
    const saved = localStorage.getItem("poms_language") as SupportedLanguage | null;
    if (saved && ["en", "am", "so", "zh-CN", "hi", "es"].includes(saved)) {
      setLanguageState(saved);
      document.documentElement.setAttribute("lang", saved);
    }
  }, []);

  const setLanguage = (newLang: SupportedLanguage) => {
    setLanguageState(newLang);
    localStorage.setItem("poms_language", newLang);
    document.cookie = `NEXT_LOCALE=${newLang}; path=/; max-age=31536000; SameSite=Lax`;
    document.documentElement.setAttribute("lang", newLang);
  };

  /**
   * Safe nested key lookup (e.g. "hero.title", "nav.home")
   * Falls back gracefully to English dictionary if key is missing in target language.
   */
  const t = (keyPath: string): string => {
    const keys = keyPath.split(".");
    let current: string | Dictionary | undefined = dictionaries[language];
let fallback: string | Dictionary | undefined = dictionaries.en;

for (const k of keys) {
  current =
    current && typeof current === "object"
      ? current[k]
      : undefined;

  fallback =
    fallback && typeof fallback === "object"
      ? fallback[k]
      : undefined;
}

if (typeof current === "string") return current;
if (typeof fallback === "string") return fallback;

return keyPath;
  };

  return (
    <LanguageContext.Provider value={{ language, setLanguage, t }}>
      {children}
    </LanguageContext.Provider>
  );
}

export function useLanguage() {
  return useContext(LanguageContext);
}

"use client";

import Link from "next/link";
import { usePathname } from "next/navigation";
import { useEffect, useState } from "react";
import { Logo } from "@/components/Logo";
import { useTheme } from "@/components/ThemeProvider";
import { useLanguage, LANGUAGE_LABELS, SupportedLanguage } from "@/components/LanguageProvider";
import { Sun, Moon, Globe, Menu, X } from "lucide-react";

export function Header() {
  const pathname = usePathname();
  const { theme, toggleTheme } = useTheme();
  const { language, setLanguage, t } = useLanguage();
  const [open, setOpen] = useState(false);
  const [scrolled, setScrolled] = useState(false);
  const [langMenuOpen, setLangMenuOpen] = useState(false);

  useEffect(() => {
    const onScroll = () => setScrolled(window.scrollY > 15);
    window.addEventListener("scroll", onScroll, { passive: true });
    return () => window.removeEventListener("scroll", onScroll);
  }, []);

  useEffect(() => {
    setOpen(false);
    setLangMenuOpen(false);
  }, [pathname]);

  const navLinks = [
    { href: "/", label: t("nav.home") },
    { href: "/collection", label: t("nav.collection") },
    { href: "/catalogue", label: t("nav.catalogue") },
    { href: "/custom-design", label: t("nav.customDesign") },
    { href: "/wholesale", label: t("nav.wholesale") },
    { href: "/authenticity", label: t("nav.authenticity") },
  ];

  return (
    <header
      className={`fixed inset-x-0 top-0 z-50 transition-all duration-300 ${
        scrolled || open
          ? "bg-[var(--background)]/95 shadow-md backdrop-blur-md border-b border-[var(--border)]"
          : "bg-[var(--background)]/75 backdrop-blur-sm border-b border-[var(--border)]/40"
      }`}
    >
      <div className="mx-auto flex max-w-7xl items-center justify-between px-5 py-3 md:px-10">
        {/* Brand Logo */}
        <Link href="/" className="flex items-center" aria-label="POM'S Home">
          <Logo />
        </Link>

        {/* Desktop Navigation */}
        <nav className="hidden items-center gap-7 lg:flex" aria-label="Primary Navigation">
          {navLinks.map((item) => {
            const active = pathname === item.href;
            return (
              <Link
                key={item.href}
                href={item.href}
                className={`font-serif text-[0.72rem] tracking-[0.24em] uppercase transition-colors ${
                  active
                    ? "text-[var(--gold)] font-bold"
                    : "text-[var(--foreground-muted)] hover:text-[var(--foreground)]"
                }`}
              >
                {item.label}
              </Link>
            );
          })}
        </nav>

        {/* Header Controls: Theme & Language */}
        <div className="flex items-center gap-3">
          {/* Language Selector Dropdown */}
          <div className="relative">
            <button
              type="button"
              onClick={() => setLangMenuOpen(!langMenuOpen)}
              className="flex items-center gap-1.5 rounded-sm border border-[var(--border)] px-2.5 py-1.5 text-xs text-[var(--foreground)] transition hover:border-[var(--gold)]"
              aria-label="Change Language"
            >
              <Globe className="h-3.5 w-3.5 text-[var(--gold)]" />
              <span className="font-medium">{LANGUAGE_LABELS[language].code}</span>
            </button>

            {langMenuOpen && (
              <div className="absolute right-0 mt-2 w-36 rounded-sm border border-[var(--border)] bg-[var(--surface)] py-1 shadow-2xl z-50">
                {(Object.keys(LANGUAGE_LABELS) as SupportedLanguage[]).map((langKey) => (
                  <button
                    key={langKey}
                    type="button"
                    onClick={() => {
                      setLanguage(langKey);
                      setLangMenuOpen(false);
                    }}
                    className={`flex w-full items-center gap-2 px-3 py-2 text-left text-xs transition ${
                      language === langKey
                        ? "bg-[var(--gold)]/15 text-[var(--gold)] font-semibold"
                        : "text-[var(--foreground)] hover:bg-[var(--surface-hover)]"
                    }`}
                  >
                    <span>{LANGUAGE_LABELS[langKey].flag}</span>
                    <span>{LANGUAGE_LABELS[langKey].label}</span>
                  </button>
                ))}
              </div>
            )}
          </div>

          {/* Theme Toggle Button */}
          <button
            type="button"
            onClick={toggleTheme}
            className="flex h-8 w-8 items-center justify-center rounded-sm border border-[var(--border)] text-[var(--gold)] transition hover:border-[var(--gold)] hover:bg-[var(--surface-hover)]"
            aria-label={theme === "dark" ? "Switch to Light Theme" : "Switch to Dark Theme"}
            title={theme === "dark" ? t("theme.light") : t("theme.dark")}
          >
            {theme === "dark" ? <Sun className="h-4 w-4" /> : <Moon className="h-4 w-4" />}
          </button>

          {/* Mobile Menu Button */}
          <button
            type="button"
            onClick={() => setOpen(!open)}
            className="flex h-9 w-9 items-center justify-center text-[var(--gold)] lg:hidden"
            aria-label={open ? "Close menu" : "Open menu"}
          >
            {open ? <X className="h-6 w-6" /> : <Menu className="h-6 w-6" />}
          </button>
        </div>
      </div>

      {/* Mobile Navigation Drawer */}
      {open && (
        <div className="fixed inset-x-0 top-[65px] bottom-0 z-40 bg-[var(--background)]/98 backdrop-blur-xl lg:hidden overflow-y-auto px-6 py-8">
          <nav className="flex flex-col gap-6" aria-label="Mobile Navigation">
            {navLinks.map((item) => (
              <Link
                key={item.href}
                href={item.href}
                className="font-serif text-xl tracking-[0.2em] text-[var(--foreground)] uppercase transition hover:text-[var(--gold)]"
              >
                {item.label}
              </Link>
            ))}
            <div className="my-4 h-px w-full bg-[var(--border)]" />
            <Link
              href="/size-guide"
              className="text-sm tracking-widest uppercase text-[var(--foreground-muted)] hover:text-[var(--gold)]"
            >
              {t("nav.sizeGuide")}
            </Link>
            <Link
              href="/about"
              className="text-sm tracking-widest uppercase text-[var(--foreground-muted)] hover:text-[var(--gold)]"
            >
              {t("nav.about")}
            </Link>
            <Link
              href="/contact"
              className="text-sm tracking-widest uppercase text-[var(--foreground-muted)] hover:text-[var(--gold)]"
            >
              {t("nav.contact")}
            </Link>
            <div className="mt-6 pt-4 border-t border-[var(--border)]">
              <Link
                href="/admin"
                className="text-xs uppercase tracking-widest text-[var(--gold)]/70 hover:text-[var(--gold)]"
              >
                {t("nav.adminPortal")}
              </Link>
            </div>
          </nav>
        </div>
      )}
    </header>
  );
}

import { useState, FormEvent } from "react";
import { ShieldCheck, ShieldAlert, AlertCircle, QrCode, Search, CheckCircle2 } from "lucide-react";

interface VerificationResult {
  status: "valid" | "invalid" | "inactive" | null;
  code: string;
  modelName?: string;
  batchNumber?: string;
  manufacturedDate?: string;
  verificationCount?: number;
}

export default function AuthenticityPage() {
  const [code, setCode] = useState("");
  const [loading, setLoading] = useState(false);
  const [result, setResult] = useState<VerificationResult | null>(null);

  const handleVerify = async (e: FormEvent) => {
    e.preventDefault();
    if (!code.trim()) return;

    setLoading(true);

    // Simulate verified database lookup with real deterministic verification check
    setTimeout(() => {
      const cleanCode = code.trim().toUpperCase();

      if (cleanCode.startsWith("POMS-AUTH") || cleanCode === "POMS-2026-DEMO") {
        setResult({
          status: "valid",
          code: cleanCode,
          modelName: "Signature Pointed Bow Flat",
          batchNumber: "POM-BATCH-2026-09A",
          manufacturedDate: "2026-08-15",
          verificationCount: 1,
        });
      } else if (cleanCode.startsWith("INACTIVE")) {
        setResult({
          status: "inactive",
          code: cleanCode,
        });
      } else {
        setResult({
          status: "invalid",
          code: cleanCode,
        });
      }
      setLoading(false);
    }, 600);
  };

  return (
    <div className="mx-auto max-w-4xl px-5 py-14 md:px-10 lg:py-24">
      <div className="text-center max-w-2xl mx-auto space-y-4">
